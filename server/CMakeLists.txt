cmake_minimum_required(VERSION 3.20)
project(credlet_server VERSION 0.3.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET libsodium sqlite3 json-c libevent libevent_extra)
add_executable(credlet-server src/main.c src/store.c src/protocol.c)
target_link_libraries(credlet-server PRIVATE PkgConfig::DEPS)
if(WIN32)
  target_link_libraries(credlet-server PRIVATE ws2_32 advapi32)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(credlet-server PRIVATE -Wall -Wextra -Wpedantic -Werror -Wformat=2 -Wshadow -fstack-protector-strong)
endif()
option(CREDLET_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(CREDLET_SANITIZE)
  target_compile_options(credlet-server PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(credlet-server PRIVATE -fsanitize=address,undefined)
endif()
install(TARGETS credlet-server RUNTIME DESTINATION bin)
